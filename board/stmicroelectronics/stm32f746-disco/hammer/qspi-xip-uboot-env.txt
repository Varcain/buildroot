baudrate=115200
bootdelay=3
bootargs=console=ttySTM0,115200 earlyprintk consoleblank=0 ignore_loglevel clk_ignore_unused root=mtd:rootfs rootfstype=cramfs ro rootwait
bootcmd=run qspi_xip_boot
qspi_xip_boot=sf probe 0:0 && sf read ${fdt_addr_r} 0xe0000 0x10000 && setenv qspi_leave_mapped 1 && sf read ${qspi_prime_r} 0 0x40 && bootm 0x900fffc0 - ${fdt_addr_r}
fdt_addr_r=0xc0408000
fdt_high=0xc0007fff
qspi_prime_r=0xc0400000
